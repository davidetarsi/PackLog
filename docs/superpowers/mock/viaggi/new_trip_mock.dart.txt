DateTime _day(int d) {
  final n = DateTime.now();
  return DateTime(n.year, n.month, n.day + d, 9);
}

String _d(BuildContext context, DateTime d, [String p = 'EEE d MMM']) =>
    DateFormat(p, context.locale.toString()).format(d);

TripModel _mockTrip({bool leg = true}) => TripModel(
  id: 'mock',
  name: 'Lisbona',
  departureDateTime: _day(9),
  returnDateTime: _day(14),
  legs: [
    if (leg)
      TripLeg(
        id: 'sintra',
        locationDisplayName: 'Sintra, Portogallo',
        from: _day(12),
        to: _day(14),
      ),
  ],
  createdAt: DateTime.now(),
  updatedAt: DateTime.now(),
);

/// Indicatore a due voci, neutro.
class _Steps extends StatelessWidget {
  final int current;
  final String second;
  const _Steps({required this.current, this.second = 'Valigia'});

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    Widget step(int i, String label) {
      final active = i == current;
      final done = i < current;
      return Row(
        mainAxisSize: MainAxisSize.min,
        children: [
          Container(
            width: context.responsive(22),
            height: context.responsive(22),
            alignment: Alignment.center,
            decoration: BoxDecoration(
              shape: BoxShape.circle,
              color: active || done ? cs.onSurface : null,
              border: Border.all(
                color: active || done ? cs.onSurface : cs.outline,
                width: 1.5,
              ),
            ),
            child: done
                ? Icon(Icons.check, size: context.iconSizeSm, color: cs.surface)
                : Text(
                    '${i + 1}',
                    style: tt.labelMedium?.copyWith(
                      color: active ? cs.surface : cs.onSurfaceVariant,
                      fontWeight: FontWeight.w700,
                    ),
                  ),
          ),
          SizedBox(width: context.spacingSm),
          Text(
            label,
            style: tt.labelLarge?.copyWith(
              color: active ? cs.onSurface : cs.onSurfaceVariant,
              fontWeight: active ? FontWeight.w700 : null,
            ),
          ),
        ],
      );
    }

    return Row(
      children: [
        step(0, 'Viaggio'),
        Expanded(
          child: Padding(
            padding: EdgeInsets.symmetric(horizontal: context.spacingSm),
            child: Divider(color: cs.outlineVariant),
          ),
        ),
        step(1, second),
      ],
    );
  }
}

Widget _sectionTitle(BuildContext context, String text) => Padding(
  padding: EdgeInsets.only(top: context.spacingLg, bottom: context.spacingSm),
  child: Text(
    text,
    style: Theme.of(
      context,
    ).textTheme.titleLarge?.copyWith(fontWeight: FontWeight.w800),
  ),
);

/// Campo "finto" con icona, testo o segnaposto, e azione a destra.
class _Field extends StatelessWidget {
  final IconData icon;
  final String? value;
  final String hint;
  final Widget? trailing;
  final bool focused;
  final String? error;
  const _Field({
    required this.icon,
    required this.hint,
    this.value,
    this.trailing,
    this.focused = false,
    this.error,
  });

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final box = Container(
      padding: EdgeInsets.symmetric(
        horizontal: context.spacingMd,
        vertical: context.spacingMd,
      ),
      decoration: BoxDecoration(
        color: context.groupSurface,
        borderRadius: BorderRadius.circular(AppConstants.inputBorderRadius),
        border: Border.all(
          color: error != null
              ? cs.error
              : focused
              ? cs.onSurface
              : cs.outlineVariant,
          width: focused || error != null ? 1.5 : 1,
        ),
      ),
      child: Row(
        children: [
          Icon(icon, size: context.iconSizeMd, color: cs.onSurfaceVariant),
          SizedBox(width: context.spacingMd),
          Expanded(
            child: Text(
              value ?? hint,
              style: value == null
                  ? tt.bodyLarge?.copyWith(color: cs.onSurfaceVariant)
                  : tt.bodyLarge?.copyWith(fontWeight: FontWeight.w600),
            ),
          ),
          ?trailing,
        ],
      ),
    );
    if (error == null) return box;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        box,
        Padding(
          padding: EdgeInsets.only(
            top: context.spacingXs,
            left: context.spacingMd,
          ),
          child: Text(error!, style: tt.bodySmall?.copyWith(color: cs.error)),
        ),
      ],
    );
  }
}

/// Meteo per tappa. `state`: meteo | meteo-sheet | clima | parziale | ...
class _WeatherSection extends StatelessWidget {
  final String state;
  const _WeatherSection({required this.state});

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final clima = state == 'clima';
    final parziale = state == 'parziale';

    Widget row(
      String city,
      String temps,
      String rain,
      IconData icon, {
      bool missing = false,
    }) => Padding(
      padding: EdgeInsets.symmetric(
        horizontal: context.spacingMd,
        vertical: context.spacingSm + context.spacingXs,
      ),
      child: Row(
        children: [
          Icon(
            missing ? Icons.cloud_off_outlined : icon,
            size: context.iconSizeMd,
            color: cs.onSurfaceVariant,
          ),
          SizedBox(width: context.spacingMd),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  city,
                  style: tt.titleSmall?.copyWith(fontWeight: FontWeight.w700),
                ),
                Text(
                  missing ? 'Meteo non disponibile per ora' : rain,
                  style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
                ),
              ],
            ),
          ),
          if (!missing)
            Text(
              temps,
              style: tt.titleMedium?.copyWith(fontWeight: FontWeight.w700),
            ),
          if (!clima && !missing) ...[
            SizedBox(width: context.spacingXs),
            Icon(
              Icons.chevron_right,
              size: context.iconSizeSm,
              color: cs.onSurfaceVariant,
            ),
          ],
        ],
      ),
    );

    return Column(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        _sectionTitle(context, 'Meteo'),
        Container(
          decoration: BoxDecoration(
            color: context.groupSurface,
            borderRadius: BorderRadius.circular(AppConstants.cardBorderRadius),
          ),
          child: Column(
            children: [
              row(
                'Lisbona · 3 notti',
                '16° – 23°',
                clima
                    ? 'Di solito piove 1 giorno su 3'
                    : 'Pioggia prevista 1 giorno su 3',
                Icons.wb_sunny_outlined,
              ),
              Divider(
                height: 1,
                indent: context.spacingMd,
                endIndent: context.spacingMd,
                color: cs.outlineVariant,
              ),
              row(
                'Sintra · 2 notti',
                '12° – 19°',
                clima
                    ? 'Di solito piove 1 giorno su 2'
                    : 'Niente pioggia prevista',
                Icons.cloud_outlined,
                missing: parziale,
              ),
            ],
          ),
        ),
        Padding(
          padding: EdgeInsets.only(
            top: context.spacingXs,
            left: context.spacingMd,
            right: context.spacingMd,
          ),
          child: Text(
            clima
                ? 'Media degli ultimi 5 anni in questi giorni. Le previsioni arrivano dal 15 ott.'
                : 'Previsioni di oggi, minima e massima medie. Tocca una tappa per i giorni.',
            style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
          ),
        ),
      ],
    );
  }
}
