const kTripsMock = String.fromEnvironment('MOCK_TRIPS', defaultValue: 'v1');

enum MockTripTab { upcoming, past, saved }

extension on MockTripTab {
  String get label => switch (this) {
    MockTripTab.upcoming => 'Prossimi',
    MockTripTab.past => 'Passati',
    MockTripTab.saved => 'Salvati',
  };
}

Color _cardColor(BuildContext context) {
  final cs = Theme.of(context).colorScheme;
  return Theme.of(context).brightness == Brightness.light
      ? cs.surfaceContainerLowest
      : cs.surfaceContainer;
}

String _destination(WidgetRef ref, TripModel trip) {
  final house = ref
      .watch(houseNotifierProvider)
      .valueOrNull
      ?.where((h) => h.id == trip.destinationHouseId)
      .firstOrNull;
  return house?.displayName ?? trip.destinationDisplayName ?? trip.name;
}

String _city(String destination) => destination.split(',').first.trim();
String? _country(String destination) {
  final parts = destination.split(',');
  return parts.length > 1 ? parts.sublist(1).join(',').trim() : null;
}

String _d(BuildContext context, DateTime? d, [String pattern = 'd MMM']) =>
    d == null ? '—' : DateFormat(pattern, context.locale.toString()).format(d);

String _countdown(TripModel trip) {
  final now = DateTime.now();
  final today = DateTime(now.year, now.month, now.day);
  if (trip.isActive) {
    final r = trip.returnDateTime;
    if (r == null) return 'In viaggio';
    final left = DateTime(r.year, r.month, r.day).difference(today).inDays;
    return left <= 0
        ? 'In viaggio · ultimo giorno'
        : 'In viaggio · ancora $left giorni';
  }
  final dep = trip.departureDateTime;
  if (dep == null) return 'Senza date';
  final days = DateTime(dep.year, dep.month, dep.day).difference(today).inDays;
  if (days == 1) return 'Domani';
  return 'Tra $days giorni';
}

({String label, IconData icon}) _cta(TripModel trip) {
  if (trip.isActive) return (label: 'Apri la checklist', icon: Icons.checklist);
  if (trip.items.isEmpty) {
    return (label: 'Prepara la valigia', icon: Icons.auto_awesome);
  }
  return (label: 'Continua a preparare', icon: Icons.luggage_outlined);
}

class _PageDots extends StatelessWidget {
  final int count;
  final int index;
  const _PageDots({required this.count, required this.index});

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    return Row(
      mainAxisAlignment: MainAxisAlignment.center,
      children: [
        for (var i = 0; i < count; i++)
          AnimatedContainer(
            duration: const Duration(milliseconds: 200),
            margin: EdgeInsets.symmetric(horizontal: context.spacingXs / 2),
            width: i == index ? context.spacingMd : context.spacingSm,
            height: context.spacingSm,
            decoration: BoxDecoration(
              color: i == index ? cs.onSurface : cs.outlineVariant,
              borderRadius: BorderRadius.circular(context.spacingXs),

          '${index + 1} di $count',
          style: Theme.of(
            context,
          ).textTheme.labelMedium?.copyWith(color: cs.onSurfaceVariant),
        ),
      ],
    );
  }
}

/// Lo staccato del biglietto: tacche nel colore della pagina e linea a
/// trattini.
class _Perforation extends StatelessWidget {
  final Axis axis;
  const _Perforation({this.axis = Axis.horizontal});

  @override
  Widget build(BuildContext context) {
    final page = Theme.of(context).scaffoldBackgroundColor;
    final dash = Theme.of(context).colorScheme.outlineVariant;
    const r = 12.0;
    Widget notch() => Container(
      width: r * 2,
      height: r * 2,
      decoration: BoxDecoration(color: page, shape: BoxShape.circle),
    );
    if (axis == Axis.vertical) {
      return SizedBox(
        width: r * 2,
        child: Stack(
          clipBehavior: Clip.none,
          children: [
            Positioned(top: -r, child: notch()),
            Positioned(bottom: -r, child: notch()),
            Positioned.fill(
              top: r,
              bottom: r,
              child: CustomPaint(painter: _Dashes(dash, vertical: true)),
            ),
          ],
        ),
      );
    }
    return SizedBox(
      height: r * 2,
      child: Stack(
        clipBehavior: Clip.none,
        children: [
          Positioned(left: -r, child: notch()),
          Positioned(right: -r, child: notch()),
          Positioned.fill(
            left: r * 1.5,
            right: r * 1.5,
            child: CustomPaint(painter: _Dashes(dash)),
          ),
        ],
      ),
    );
  }
}

class _Dashes extends CustomPainter {
  final Color color;
  final bool vertical;
  _Dashes(this.color, {this.vertical = false});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..strokeWidth = 1.5;
    const dash = 6.0, gap = 5.0;
    if (vertical) {
      final x = size.width / 2;
      for (var y = 0.0; y < size.height; y += dash + gap) {
        canvas.drawLine(Offset(x, y), Offset(x, y + dash), paint);
      }
    } else {
      final y = size.height / 2;
      for (var x = 0.0; x < size.width; x += dash + gap) {
        canvas.drawLine(Offset(x, y), Offset(x + dash, y), paint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant _Dashes old) => old.color != color;
}

/// Cosa manca ancora in valigia: i primi tre oggetti non spuntati.
class _Missing extends StatelessWidget {
  final TripModel trip;
  const _Missing({required this.trip});

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final missing = trip.items.where((i) => !i.isChecked).toList();
    if (missing.isEmpty) return const SizedBox.shrink();
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          'DA METTERE IN VALIGIA',
          style: tt.labelSmall?.copyWith(
            color: cs.onSurfaceVariant,
            letterSpacing: 1.2,
          ),
        ),
        SizedBox(height: context.spacingSm),
        for (final i in missing.take(3))
          Padding(
            padding: EdgeInsets.only(bottom: context.spacingXs),
            child: Row(
              children: [
                Icon(
                  Icons.check_box_outline_blank,
                  size: context.iconSizeSm,
                  color: cs.onSurfaceVariant,
                ),
                SizedBox(width: context.spacingSm),
                Expanded(
                  child: Text(
                    i.name,
                    maxLines: 1,
                    overflow: TextOverflow.ellipsis,
                    style: tt.bodyMedium,
                  ),
                ),
                if (i.quantity > 1)
                  Text(
                    'x${i.quantity}',
                    style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
                  ),
              ],
            ),
          ),
        if (missing.length > 3)
          Text(
            'e altri ${missing.length - 3}',
            style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
          ),
      ],
    );
  }
}

class _Progress extends StatelessWidget {
  final TripModel trip;
  const _Progress({required this.trip});

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final total = trip.totalCount;
    final done = trip.completedCount;
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(
          total == 0 ? 'Valigia ancora vuota' : '$done di $total in valigia',
          style: tt.labelLarge,
        ),
        SizedBox(height: context.spacingXs),
        ClipRRect(
          borderRadius: BorderRadius.circular(context.spacingXs),
          child: LinearProgressIndicator(
            value: total == 0 ? 0 : done / total,
            minHeight: context.spacingXs + 2,
            backgroundColor: cs.surfaceContainerHigh,
            color: cs.onSurface,
          ),
        ),
      ],
    );
  }
}

    final byYear = <int, List<TripModel>>{};
    for (final t in trips) {
      byYear
          .putIfAbsent((t.departureDateTime ?? t.createdAt).year, () => [])
          .add(t);
    }
    return ListView(
      padding: EdgeInsets.fromLTRB(
        context.spacingMd,
        0,
        context.spacingMd,
        context.navBarReservedHeight + context.spacingMd,
      ),
      children: [
        if (kTripsMock == 'v12')
          ..._postcardRows(context, trips, saved)
        else if (kTripsMock == 'v12b')
          ..._postcardAlbum(context, trips, saved)
        else if (kTripsMock == 'v1')
          for (final t in trips)
            Padding(
              padding: EdgeInsets.only(bottom: context.spacingSm),
              child: _StubRow(trip: t, saved: saved),
            )
        else
          for (final e in byYear.entries) ...[
            DsGroupHeader(
              label: saved ? 'Modelli' : '${e.key}',
              count: e.value.length,
            ),
            Container(
              decoration: BoxDecoration(
                color: _cardColor(context),
                borderRadius: BorderRadius.circular(
                  AppConstants.cardBorderRadius,
                ),
              ),
              child: Column(
                children: [
                  for (final (i, t) in e.value.indexed) ...[
                    if (i > 0)
                      Divider(
                        height: 1,
                        indent: context.spacingMd,
                        endIndent: context.spacingMd,
                        color: Theme.of(context).colorScheme.outlineVariant,
                      ),
                    _PlainRow(trip: t, saved: saved),
                  ],
                ],
              ),
            ),
          ],
      ],
    );
  }
}

int? _nights(TripModel trip) {
  final a = trip.departureDateTime, b = trip.returnDateTime;
  if (a == null || b == null) return null;
  return DateTime(
    b.year,
    b.month,
    b.day,
  ).difference(DateTime(a.year, a.month, a.day)).inDays;
}

String _bags(TripModel trip) => trip.luggageCount == 0

// ─────────────────────────────────────────────────────────────────────────
// V4: agenda. Il tempo al centro: una linea verticale con partenza, tappe,
// "oggi" e ritorno. La valigia è un solo riquadro riassuntivo.
// ─────────────────────────────────────────────────────────────────────────
class _TicketV4 extends ConsumerWidget {
  final TripModel trip;
  const _TicketV4({required this.trip});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final dest = _destination(ref, trip);
    final cta = _cta(trip);
    final now = DateTime.now();
    final today = DateTime(now.year, now.month, now.day);

    final stops =
        <({DateTime? date, String title, String? sub, bool today})>[
          (
            date: today,
            title: 'Oggi',
            sub: trip.isActive ? 'Sei in viaggio' : _countdown(trip),
            today: true,
          ),
          (
            date: trip.departureDateTime,
            title: 'Partenza',
            sub: trip.legs.length > 1
                ? 'per ${_city(trip.legs.first.locationDisplayName)}'
                : 'per ${_city(dest)}',
            today: false,
          ),
          if (trip.legs.length > 1)
            for (final l in trip.legs.skip(1))
              (
                date: l.from,
                title: _city(l.locationDisplayName),
                sub: 'tappa',
                today: false,
              ),
          (
            date: trip.returnDateTime,
            title: 'Ritorno',
            sub: null,
            today: false,
          ),
        ]..sort(
          (a, b) =>
              (a.date ?? DateTime(2099)).compareTo(b.date ?? DateTime(2099)),
        );
    final todayIndex = stops.indexWhere((s) => s.today);

    Widget stop(int i) {
      final s = stops[i];
      final past = i < todayIndex;
      final dot = context.responsive(s.today ? 14 : 10);
      return IntrinsicHeight(
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            SizedBox(
              width: context.responsive(16),
              child: Column(
                children: [
                  Container(
                    width: 2,
                    height: context.spacingXs,
                    color: i == 0 ? Colors.transparent : cs.outlineVariant,
                  ),
                  Container(
                    width: dot,
                    height: dot,
                    decoration: BoxDecoration(
                      shape: BoxShape.circle,
                      color: s.today || past
                          ? cs.onSurface
                          : _cardColor(context),
                      border: Border.all(
                        color: s.today || past ? cs.onSurface : cs.outline,
                        width: 2,
                      ),
                    ),
                  ),
                  Expanded(
                    child: Container(
                      width: 2,
                      color: i == stops.length - 1
                          ? Colors.transparent
                          : cs.outlineVariant,
                    ),
                  ),
                ],
              ),
            ),
            SizedBox(width: context.spacingMd),
            Expanded(
              child: Padding(
                padding: EdgeInsets.only(bottom: context.spacingLg),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      s.title,
                      style: (s.today ? tt.titleLarge : tt.titleMedium)
                          ?.copyWith(
                            fontWeight: s.today ? FontWeight.w700 : null,
                            color: past ? cs.onSurfaceVariant : null,
                          ),
                    ),
                    if (s.sub != null)
                      Text(
                        s.sub!,
                        style: tt.bodySmall?.copyWith(
                          color: cs.onSurfaceVariant,
                        ),
                      ),
                  ],
                ),
              ),
            ),
            Text(
              _d(context, s.date, 'EEE d MMM'),
              style: tt.titleSmall?.copyWith(
                color: past ? cs.onSurfaceVariant : null,
              ),
            ),
          ],
        ),
      );
    }

    final total = trip.totalCount, done = trip.completedCount;
    return GestureDetector(
      onTap: () => context.push('/trips/${trip.id}'),
      child: Container(
        decoration: _card(context),
        padding: EdgeInsets.all(context.spacingLg),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text(
              _city(dest),
              maxLines: 2,
              overflow: TextOverflow.ellipsis,
              style: tt.headlineMedium?.copyWith(fontWeight: FontWeight.w700),
            ),
            Text(
              [
                if (_country(dest) != null) _country(dest)!,
                if (trip.name != _city(dest)) trip.name,
              ].join(' · '),
              style: tt.bodyMedium?.copyWith(color: cs.onSurfaceVariant),
            ),
            SizedBox(height: context.spacingLg),
            for (var i = 0; i < stops.length; i++) stop(i),
            const Spacer(),
            Container(
              padding: EdgeInsets.all(context.spacingMd),
              decoration: BoxDecoration(
                color: cs.surfaceContainerHigh,
                borderRadius: BorderRadius.circular(context.spacingMd),
              ),
              child: Row(
                children: [
                  Icon(Icons.luggage_outlined, size: context.iconSizeMd),
                  SizedBox(width: context.spacingMd),
                  Expanded(
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Text(
                          total == 0
                              ? 'Valigia ancora vuota'
                              : '$done di $total in valigia',
                          style: tt.titleSmall,
                        ),
                        Text(
                          trip.luggageCount == 0
                              ? 'Nessun bagaglio scelto'
                              : trip.luggages.map((l) => l.name).join(' · '),
                          maxLines: 1,
                          overflow: TextOverflow.ellipsis,
                          style: tt.bodySmall?.copyWith(
                            color: cs.onSurfaceVariant,
                          ),
                        ),
                      ],
                    ),
                  ),
                  if (total > 0)
                    Text(
                      '${(done * 100 / total).round()}%',
                      style: tt.titleMedium,
                    ),
                ],
              ),
            ),
            SizedBox(height: context.spacingMd),
            DsButton(
              label: cta.label,
              icon: cta.icon,
              expand: true,
              onPressed: () => context.push('/trips/${trip.id}'),
            ),
          ],
        ),
      ),
    );
  }
}

class _Wave extends CustomPainter {
  final double level;
  final Color color;
  _Wave({required this.level, required this.color});

  @override
  void paint(Canvas canvas, Size size) {
    if (level <= 0) return;
    final y0 = size.height * (1 - level);
    const amp = 7.0;
    final wl = size.width / 1.2;
    final path = Path()..moveTo(0, y0);
    for (var x = 0.0; x <= size.width + 4; x += 4) {
      path.lineTo(x, y0 + math.sin(x / wl * 2 * math.pi) * amp);
    }
    path
      ..lineTo(size.width, size.height)
      ..lineTo(0, size.height)
      ..close();
    canvas.drawPath(path, Paint()..color = color);
  }

  @override
  bool shouldRepaint(covariant _Wave old) =>
      old.level != level || old.color != color;
}

// V7: cartoline. I prossimi viaggi sono una pila di cartoline storte; si
// scorre lanciando via quella in cima. Francobollo col conto alla rovescia,
// timbro con la data, righe dell'indirizzo con date e bagagli.
// ─────────────────────────────────────────────────────────────────────────
class _PostcardPile extends StatefulWidget {
  final List<TripModel> trips;
  final int index;
  final ValueChanged<int> onChanged;
  final Widget Function(TripModel)? builder;
  const _PostcardPile({
    required this.trips,
    required this.index,
    required this.onChanged,
    this.builder,
  });

  @override
  State<_PostcardPile> createState() => _PostcardPileState();
}

class _PostcardPileState extends State<_PostcardPile> {
  double _dx = 0;

  @override
  Widget build(BuildContext context) {
    final n = widget.trips.length;
    final i = widget.index;
    const tilts = [0.0, 0.045, -0.06];
    return GestureDetector(
      key: const Key('trips_pager'),
      behavior: HitTestBehavior.opaque,
      onHorizontalDragUpdate: (d) => setState(() => _dx += d.delta.dx),
      onHorizontalDragEnd: (_) {
        if (_dx < -60 && i < n - 1) widget.onChanged(i + 1);
        if (_dx > 60 && i > 0) widget.onChanged(i - 1);
        setState(() => _dx = 0);
      },
      child: Padding(
        padding: EdgeInsets.symmetric(
          horizontal: context.spacingLg,
          vertical: context.spacingSm,
        ),
        child: Stack(
          children: [
            for (var k = math.min(i + 2, n - 1); k >= i; k--)
              Positioned.fill(
                child: Transform.translate(
                  offset: k == i
                      ? Offset(_dx, 0)
                      : Offset(10.0 * (k - i), 8.0 * (k - i)),
                  child: Transform.rotate(
                    angle: tilts[k - i] + (k == i ? _dx / 2000 : 0),
                    child: k == i
                        ? (widget.builder ?? (t) => _Postcard(trip: t))(
                            widget.trips[k],
                          )
                        : const _BlankPostcard(),
                  ),
                ),
              ),
          ],
        ),
      ),
    );
  }
}

/// Francobollo: rettangolo con i bordi "mangiati" a semicerchi e una cornice
/// interna.
class _StampPainter extends CustomPainter {
  final Color color;
  final Color border;
  _StampPainter({required this.color, required this.border});

  @override
  void paint(Canvas canvas, Size size) {
    const r = 3.5, step = 10.0;
    var shape = Path()..addRect(Offset.zero & size);
    final bites = Path();
    for (var x = step / 2; x < size.width; x += step) {
      bites
        ..addOval(Rect.fromCircle(center: Offset(x, 0), radius: r))
        ..addOval(Rect.fromCircle(center: Offset(x, size.height), radius: r));
    }
    for (var y = step / 2; y < size.height; y += step) {
      bites
        ..addOval(Rect.fromCircle(center: Offset(0, y), radius: r))
        ..addOval(Rect.fromCircle(center: Offset(size.width, y), radius: r));
    }
    shape = Path.combine(PathOperation.difference, shape, bites);
    canvas.drawPath(shape, Paint()..color = color);
    canvas.drawRect(
      (Offset.zero & size).deflate(8),
      Paint()
        ..color = border
        ..style = PaintingStyle.stroke
        ..strokeWidth = 1,
    );
  }

  @override
  bool shouldRepaint(covariant _StampPainter old) =>
      old.color != color || old.border != border;
}

class _RoutePainter extends CustomPainter {
  final List<String> stops;
  final bool active;
  final String startLabel;
  final Color dash, ink, muted, card;
  final TextStyle stopStyle, lastStyle, smallStyle;
  _RoutePainter({
    required this.stops,
    required this.active,
    required this.startLabel,
    required this.dash,
    required this.ink,
    required this.muted,
    required this.card,
    required this.stopStyle,
    required this.lastStyle,
    required this.smallStyle,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final w = size.width, h = size.height;
    final p0 = Offset(w * 0.06, h * 0.86);
    final c = Offset(w * 0.18, h * 0.02);
    final p1 = Offset(w * 0.88, h * 0.26);
    Offset at(double t) =>
        p0 * ((1 - t) * (1 - t)) + c * (2 * (1 - t) * t) + p1 * (t * t);
    Offset tangent(double t) => (c - p0) * (2 * (1 - t)) + (p1 - c) * (2 * t);

    final path = Path()
      ..moveTo(p0.dx, p0.dy)
      ..quadraticBezierTo(c.dx, c.dy, p1.dx, p1.dy);
    final dashPaint = Paint()
      ..color = dash
      ..strokeWidth = 2
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round;
    for (final m in path.computeMetrics()) {
      for (var d = 0.0; d < m.length; d += 13) {
        canvas.drawPath(m.extractPath(d, d + 6), dashPaint);
      }
    }

    void label(
      String text,
      TextStyle style,
      Offset anchor, {
      bool above = false,
    }) {
      final tp = TextPainter(
        text: TextSpan(text: text, style: style),
        textDirection: ui.TextDirection.ltr,
        maxLines: 1,
        ellipsis: '…',
      )..layout(maxWidth: w);
      final x = (anchor.dx - tp.width / 2).clamp(0.0, w - tp.width);
      final y = above ? anchor.dy - tp.height - 12 : anchor.dy + 12;
      tp.paint(canvas, Offset(x, y));
    }

    // Partenza.
    canvas.drawCircle(p0, 5, Paint()..color = muted);
    label(startLabel, smallStyle, p0);

    // Tappe.
    final ts = stops.length == 1
        ? [1.0]
        : [
            for (var i = 0; i < stops.length; i++)
              0.62 + 0.38 * i / (stops.length - 1),
          ];
    for (var i = 0; i < stops.length; i++) {
      final pt = at(ts[i]);
      final last = i == stops.length - 1;
      canvas.drawCircle(pt, last ? 9 : 6, Paint()..color = ink);
      canvas.drawCircle(pt, last ? 4 : 2.5, Paint()..color = card);
      label(stops[i], last ? lastStyle : stopStyle, pt);
    }

    // Aereo.
    final tp = active ? ts.first - 0.1 : 0.28;
    final pt = at(tp);
    final tg = tangent(tp);
    canvas.drawCircle(pt, 18, Paint()..color = card);
    final icon = TextPainter(
      text: TextSpan(
        text: String.fromCharCode(Icons.flight.codePoint),
        style: TextStyle(
          fontFamily: Icons.flight.fontFamily,
          package: Icons.flight.fontPackage,
          fontSize: 28,
          color: ink,
        ),
      ),
      textDirection: ui.TextDirection.ltr,
    )..layout();
    canvas
      ..save()
      ..translate(pt.dx, pt.dy)
      ..rotate(math.atan2(tg.dy, tg.dx) + math.pi / 2);
    icon.paint(canvas, Offset(-icon.width / 2, -icon.height / 2));
    canvas.restore();
  }

  @override
  bool shouldRepaint(covariant _RoutePainter old) => true;
}

int? _daysLeft(TripModel trip) {
  final dep = trip.departureDateTime;
  if (dep == null) return null;
  final now = DateTime.now();
  return DateTime(
    dep.year,
    dep.month,
    dep.day,
  ).difference(DateTime(now.year, now.month, now.day)).inDays;
}

typedef _Place = ({String name, DateTime? from, DateTime? to});

List<_Place> _places(TripModel trip, String dest) => trip.legs.isNotEmpty
    ? [
        for (final l in trip.legs)
          (name: _city(l.locationDisplayName), from: l.from, to: l.to),
      ]
    : [
        (
          name: _city(dest),
          from: trip.departureDateTime,
          to: trip.returnDateTime,
        ),
      ];

String _nightsLabel(DateTime? a, DateTime? b) {
  if (a == null || b == null) return '';
  final n = DateTime(
    b.year,
    b.month,
    b.day,
  ).difference(DateTime(a.year, a.month, a.day)).inDays;
  return n == 1 ? '1 notte' : '$n notti';
}

enum _StampShape { rect, circle, oval }

class _Stamp extends StatelessWidget {
  final _StampShape shape;
  final IconData? icon;
  final String top;
  final String main;
  final String bottom;
  const _Stamp({
    required this.shape,
    this.icon,
    required this.top,
    required this.main,
    required this.bottom,
  });

  @override
  Widget build(BuildContext context) {
    final tt = Theme.of(context).textTheme;
    final ink = Theme.of(context).colorScheme.onSurface.withValues(alpha: 0.72);
    final base = context.responsive(118);
    final (w, h) = switch (shape) {
      _StampShape.rect => (base * 0.95, base * 0.72),
      _StampShape.circle => (base, base),
      _StampShape.oval => (base, base * 0.62),
    };
    BoxDecoration deco(double width) => BoxDecoration(
      shape: shape == _StampShape.circle ? BoxShape.circle : BoxShape.rectangle,
      borderRadius: switch (shape) {
        _StampShape.rect => BorderRadius.circular(context.spacingXs),
        _StampShape.oval => BorderRadius.circular(base),
        _StampShape.circle => null,
      },
      border: Border.all(color: ink, width: width),
    );
    return Container(
      width: w,
      height: h,
      padding: const EdgeInsets.all(3),
      decoration: deco(2.5),
      child: Container(
        decoration: deco(1),
        padding: EdgeInsets.symmetric(
          horizontal: shape == _StampShape.circle ? 14 : 10,
          vertical: 6,
        ),
        child: FittedBox(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              if (icon != null) Icon(icon, size: 18, color: ink),
              Text(
                top,
                style: tt.labelSmall?.copyWith(color: ink, letterSpacing: 1.5),
              ),
              Text(
                main,
                style: tt.titleLarge?.copyWith(
                  color: ink,
                  fontWeight: FontWeight.w900,
                ),
              ),
              Text(
                bottom,
                style: tt.labelSmall?.copyWith(color: ink, letterSpacing: 1),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

class _Guilloche extends CustomPainter {
  final Color color;
  _Guilloche({required this.color});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = 0.8;
    for (var y = 0.0; y < size.height + 20; y += 16) {
      final path = Path()..moveTo(0, y);
      for (var x = 0.0; x <= size.width; x += 4) {
        path.lineTo(x, y + math.sin(x / 38 + y / 50) * 4);
      }
      canvas.drawPath(path, paint);
    }
  }

  @override
  bool shouldRepaint(covariant _Guilloche old) => old.color != color;
}

Map<int, List<TripModel>> _byYear(List<TripModel> trips) {
  final m = <int, List<TripModel>>{};
  for (final t in trips) {
    m.putIfAbsent((t.departureDateTime ?? t.createdAt).year, () => []).add(t);
  }
  return m;
}

List<Widget> _grouped(
  BuildContext context,
  List<TripModel> trips,
  bool saved,
  Widget Function(Map<int, List<TripModel>>, MapEntry<int, List<TripModel>>)
  group,
) {
  final m = saved ? {0: trips} : _byYear(trips);
  return [
    for (final e in m.entries) ...[
      DsGroupHeader(
        label: saved ? 'Da riusare' : '${e.key}',
        count: e.value.length,
      ),
      group(m, e),
      SizedBox(height: context.spacingSm),
    ],
  ];
}

// ── V12 A: righe a mini cartolina (dal mini biglietto di V1). ──────────────
List<Widget> _postcardRows(
  BuildContext context,
  List<TripModel> trips,
  bool saved,
) => _grouped(
  context,
  trips,
  saved,
  (_, e) => Column(
    children: [
      for (final (i, t) in e.value.indexed)
        Padding(

          child: Transform.rotate(
            angle: i.isEven ? -0.006 : 0.006,

          ),
        ),
    ],
  ),
);

BoxDecoration _paper(BuildContext context) {
  final cs = Theme.of(context).colorScheme;
  return BoxDecoration(
    color: _cardColor(context),
    borderRadius: BorderRadius.circular(context.spacingSm),
    border: Border.all(color: cs.outlineVariant.withValues(alpha: 0.6)),
    boxShadow: [
      BoxShadow(
        color: Colors.black.withValues(alpha: 0.06),
        blurRadius: 10,
        offset: const Offset(0, 3),
      ),
    ],
  );
}

/// Francobollo piccolo con giorno e mese. Se il viaggio è passato è
/// "annullato" dalle onde del timbro postale.
class _DateStamp extends StatelessWidget {
  final TripModel trip;
  final bool cancelled;
  final double width;
  const _DateStamp({
    required this.trip,
    required this.cancelled,
    required this.width,
  });

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final d = trip.departureDateTime;
    return SizedBox(
      width: width * 1.5,
      height: width * 1.2,
      child: Stack(
        children: [
          CustomPaint(
            painter: _StampPainter(
              color: cs.surfaceContainerHigh,
              border: cs.outlineVariant,
            ),
            child: SizedBox(
              width: width,
              height: width * 1.2,
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Text(
                    _d(context, d, 'd'),
                    style: tt.titleLarge?.copyWith(fontWeight: FontWeight.w800),
                  ),
                  Text(
                    _d(context, d, 'MMM yy').toUpperCase(),
                    style: tt.labelSmall?.copyWith(color: cs.onSurfaceVariant),
                  ),
                ],
              ),
            ),
          ),
          if (cancelled)
            Positioned.fill(
              left: width * 0.78,
              child: CustomPaint(
                painter: _CancelWaves(
                  color: cs.onSurfaceVariant.withValues(alpha: 0.55),
                ),
              ),
            ),
        ],
      ),
    );
  }
}

/// Le onde del timbro postale che "annullano" il francobollo.
class _CancelWaves extends CustomPainter {
  final Color color;
  _CancelWaves({required this.color});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.4;
    for (var k = 0; k < 4; k++) {
      final y = size.height * (0.3 + 0.13 * k);
      final path = Path()..moveTo(0, y);
      for (var x = 0.0; x <= size.width; x += 2) {
        path.lineTo(x, y + math.sin(x / 5) * 2.2);
      }
      canvas.drawPath(path, paint);
    }
  }

  @override
  bool shouldRepaint(covariant _CancelWaves old) => old.color != color;
}

Widget _reuseButton(BuildContext context) => TextButton.icon(

    foregroundColor: Theme.of(context).colorScheme.onSurface,
    visualDensity: VisualDensity.compact,

);

class _PostcardRow extends ConsumerWidget {
  final TripModel trip;
  final bool saved;
  const _PostcardRow({required this.trip, required this.saved});

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final cs = Theme.of(context).colorScheme;
    final tt = Theme.of(context).textTheme;
    final dest = _destination(ref, trip);
    return Container(
      padding: EdgeInsets.fromLTRB(
        context.spacingMd,
        context.spacingSm,
        context.spacingXs,
        context.spacingSm,
      ),
      decoration: _paper(context),
      child: Row(
        children: [
          _DateStamp(
            trip: trip,
            cancelled: !saved,
            width: context.responsive(50),
          ),
          SizedBox(width: context.spacingSm),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  'Saluti da',
                  style: tt.labelMedium?.copyWith(
                    fontStyle: FontStyle.italic,
                    color: cs.onSurfaceVariant,
                  ),
                ),
                Text(
                  _city(dest),
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                  style: tt.titleMedium?.copyWith(fontWeight: FontWeight.w800),
                ),
                Text(
                  [
                    if (trip.name != _city(dest)) trip.name,
                    '${_d(context, trip.departureDateTime)} – ${_d(context, trip.returnDateTime)}',
                  ].join(' · '),
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                  style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
                ),
              ],
            ),
          ),
          _reuseButton(context),
        ],
      ),
    );
  }
}

// ── V12 B: album di cartoline, due per riga, appese un po' storte. ────────
List<Widget> _postcardAlbum(
  BuildContext context,
  List<TripModel> trips,
  bool saved,
) => _grouped(
  context,
  trips,
  saved,
  (_, e) => GridView.count(
    crossAxisCount: 2,
    shrinkWrap: true,
    physics: const NeverScrollableScrollPhysics(),
    mainAxisSpacing: context.spacingMd,
    crossAxisSpacing: context.spacingMd,
    childAspectRatio: 0.82,
    children: [
      for (final (i, t) in e.value.indexed)
        Transform.rotate(
          angle: const [-0.025, 0.02, 0.012, -0.018][i % 4],
          child: _AlbumPostcard(trip: t, saved: saved),
        ),
    ],
  ),
);
